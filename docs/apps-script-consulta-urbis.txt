/* ═══════════════════════════════════════════════════════════════════════════
   URBIS · CONSULTA CIUDADANA (aprobación presidencial)
   ───────────────────────────────────────────────────────────────────────────
   QUÉ HACE
   Guarda los votos de la consulta del módulo Seguimiento Presidencial y
   devuelve el conteo. Un voto por dispositivo: si el mismo aparato vuelve a
   votar, se ACTUALIZA su voto en vez de sumar uno nuevo.

   CÓMO SE INSTALA (2 pasos)
   1) Abre tu Apps Script de URBIS (el mismo de docs/apps-script-urbis-COMPLETO.txt).
   2) Pega TODO este archivo al final, y añade estas dos líneas dentro de
      doPost, junto a las demás `if (action === ...)`:

        if (action === 'consulta_voto')       return json_(consultaVoto_(body));
        if (action === 'consulta_resultados') return json_(consultaResultados_(body));

   Luego: Implementar → Administrar implementaciones → editar → Nueva versión.
   La URL NO cambia, así que no hay que tocar nada en la app.

   MIENTRAS NO SE INSTALE: la app sigue funcionando. El voto se guarda en el
   propio teléfono y la pantalla avisa de que el total aún no se sincroniza.

   PRIVACIDAD: no se guarda usuario, correo ni IP. El campo `dispositivo` es
   un número aleatorio que genera el navegador y que solo sirve para no contar
   dos veces el mismo aparato.
   ═══════════════════════════════════════════════════════════════════════════ */

var URBIS_CONSULTA_HOJA = 'consulta_urbis';

/* Registra o actualiza el voto de un dispositivo. */
function consultaVoto_(body) {
  var consulta   = String((body && body.consulta) || '').trim();
  var opcion     = String((body && body.opcion) || '').trim();
  var dispositivo= String((body && body.dispositivo) || '').trim();

  if (!consulta || !opcion || !dispositivo) {
    return { ok: false, error: 'Faltan datos de la consulta.' };
  }
  // Lista blanca: evita que alguien inyecte opciones inventadas por POST.
  if (['aprueba', 'desaprueba', 'nsnr'].indexOf(opcion) === -1) {
    return { ok: false, error: 'Opción no válida.' };
  }

  var sh = sheet_(URBIS_CONSULTA_HOJA, ['consulta', 'dispositivo', 'opcion', 'fecha']);
  var lock = LockService.getScriptLock();
  try {
    // Sin el candado, dos votos simultáneos pueden leer la misma última fila
    // y duplicar al mismo dispositivo.
    lock.waitLock(8000);

    var datos = sh.getDataRange().getValues();
    var fila = -1;
    for (var i = 1; i < datos.length; i++) {
      if (String(datos[i][0]) === consulta && String(datos[i][1]) === dispositivo) { fila = i + 1; break; }
    }
    var ahora = new Date().toISOString();
    if (fila > 0) {
      sh.getRange(fila, 3).setValue(opcion);
      sh.getRange(fila, 4).setValue(ahora);
    } else {
      sh.appendRow([consulta, dispositivo, opcion, ahora]);
    }
  } catch (err) {
    return { ok: false, error: String(err) };
  } finally {
    try { lock.releaseLock(); } catch (e) {}
  }

  return consultaResultados_({ consulta: consulta });
}

/* Devuelve el conteo por opción: { ok:true, conteo:{aprueba:N,...}, total:N } */
function consultaResultados_(body) {
  var consulta = String((body && body.consulta) || '').trim();
  if (!consulta) return { ok: false, error: 'Falta el id de la consulta.' };

  var sh = sheet_(URBIS_CONSULTA_HOJA, ['consulta', 'dispositivo', 'opcion', 'fecha']);
  var datos = sh.getDataRange().getValues();
  var conteo = { aprueba: 0, desaprueba: 0, nsnr: 0 };
  var total = 0;

  for (var i = 1; i < datos.length; i++) {
    if (String(datos[i][0]) !== consulta) continue;
    var op = String(datos[i][2] || '').trim();
    if (conteo.hasOwnProperty(op)) { conteo[op]++; total++; }
  }
  return { ok: true, consulta: consulta, conteo: conteo, total: total };
}
