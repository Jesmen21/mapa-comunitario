 *    Con este script la contraseña queda GUARDADA COMO HASH en tu hoja de

 * --------------------------------------
 * 1. Abre tu proyecto de Apps Script (el mismo del archivo

 * 3. Arriba, en el selector de funciones, elige  crearCuentasSistemaUrbis
 * 4. ANTES de ejecutar, cambia las dos contraseñas de abajo (marcadas con
 *    CAMBIA_ESTO). Escríbelas aquí, en el editor: este archivo vive en tu
 *    cuenta de Google, no en el repositorio público.
 * 5. Pulsa Ejecutar. Verás en el registro (Ver > Registro de ejecución) el
 *    resultado de cada cuenta.
 * 6. Cuando confirmes que puedes entrar en la app con esas cuentas, avísame:
 *    borro el bloque ADMIN de js/00-config.js y el atajo del cliente, y con eso
 *    las contraseñas dejan de estar publicadas.
 * 7. IMPORTANTE: después de ejecutar, vuelve a poner CAMBIA_ESTO en el lugar de
 *    las contraseñas y guarda. Así no quedan escritas ni siquiera en el editor.
 *
 * NOTA DE SEGURIDAD (léela)
 * -------------------------
 * Esto elimina la publicación de las contraseñas, que es el agujero más grave.
 * Pero las acciones de administrador (borrar reportes, dar permisos) todavía se
 * autorizan en el navegador, no en el servidor. Alguien con conocimientos puede
 * saltárselas editando el JavaScript en SU propio navegador. Cerrar eso del todo
 * es un trabajo aparte: hay que exigir un token de sesión en cada acción de
 * escritura del Apps Script. Queda anotado como pendiente.

function crearCuentasSistemaUrbis() {
  // ── EDITA AQUÍ ────────────────────────────────────────────────────────────
  // Usa contraseñas largas (12+ caracteres). No reutilices las de otros sitios.
  var CUENTAS = [
    { usuario: 'urbisnoticia', nombre: 'URBIS Noticias', password: 'CAMBIA_ESTO_1' },
    { usuario: 'urbisadmin',   nombre: 'Administrador URBIS', password: 'CAMBIA_ESTO_2' }
  ];
  // ──────────────────────────────────────────────────────────────────────────

  CUENTAS.forEach(function (c) {
    if (String(c.password).indexOf('CAMBIA_ESTO') === 0) {
      Logger.log('SALTADA "' + c.usuario + '": todavía tiene la contraseña de ejemplo.');
      return;
    }
    if (String(c.password).length < 8) {
      Logger.log('SALTADA "' + c.usuario + '": la contraseña es demasiado corta.');
      return;
    }
    var r = upsertCuentaSistemaUrbis_(c.usuario, c.nombre, c.password);
    Logger.log(r.creada
      ? ('CREADA "' + r.usuario + '" (fila ' + r.fila + ')')
      : ('ACTUALIZADA "' + r.usuario + '" (fila ' + r.fila + ') — contraseña cambiada'));
  });
  Logger.log('Listo. Ahora entra en la app con el usuario y la contraseña que pusiste.');
}

/**
 * Crea o actualiza una cuenta de sistema en la hoja de usuarios.
 * Queda PRE-VERIFICADA (no pide código de correo) y con rol admin, así entra
 * por el mismo loginUser_() que el resto — sin atajos en el navegador.
 */
function upsertCuentaSistemaUrbis_(usuarioRaw, nombreRaw, passwordRaw) {
  var usuario = normUser_(usuarioRaw);
  var nombre = String(nombreRaw || usuario);
  var password = String(passwordRaw || '');

  var sh = sheet_(URBIS_AUTH.SHEET_USERS, baseHeaders_().users);
  var valores = sh.getDataRange().getValues();
  var encabezados = valores[0].map(function (h) { return String(h || '').trim(); });
  var colUsuario = encabezados.indexOf('usuario');
  if (colUsuario < 0) throw new Error('La hoja de usuarios no tiene la columna "usuario".');

  // ¿Ya existe?
  var fila = -1;
  for (var i = 1; i < valores.length; i++) {
    if (normUser_(valores[i][colUsuario]) === usuario) { fila = i + 1; break; }
  }

  var creada = false;
  if (fila < 0) {
    sh.appendRow([]);
    fila = sh.getLastRow();
    creada = true;
  }

  // El salt es propio de la cuenta y estable, para que loginUser_() lo reproduzca.
  var userId = 'sys_' + usuario;
  var salt = userId;

  var campos = {
    user_id: userId,
    usuario: usuario,
    nombres: nombre,
    apellidos: '',
    nombre_completo: nombre,
    correo: '',                    // sin correo: es una cuenta de sistema
    rol_solicitado: 'admin',
    estado_cuenta: 'activo',       // isVerified_() acepta estado_cuenta = activo
    email_verificado: 'si',        // ...y también email_verificado = si
    password_hash: hashPassword_(password, salt),
    password_salt: salt,
    codigo_verificacion: '',
    codigo_expira_en: '',
    acepta_terminos: 'si',
    observaciones_admin: 'Cuenta de sistema URBIS (creada con crearCuentasSistemaUrbis)'
  };
  if (creada) campos.fecha_registro = nowIso_();
  campos.fecha_verificacion = nowIso_();

  Object.keys(campos).forEach(function (h) {
    if (encabezados.indexOf(h) >= 0) setCellByHeader_(sh, fila, h, campos[h]);
  });

  return { usuario: usuario, fila: fila, creada: creada };
}

/**
 * Comprobación opcional: verifica que la contraseña quedó bien guardada,
 * SIN exponerla. Devuelve solo true/false en el registro.
 *
 * Cambia los dos valores, ejecuta, mira el registro y vuelve a dejarlos vacíos.
 */
function probarCuentaSistemaUrbis() {
  var usuario = '';   // ej: 'urbisnoticia'
  var password = '';  // la que acabas de poner
  if (!usuario || !password) { Logger.log('Escribe usuario y contraseña dentro de la función para probar.'); return; }
  var out = loginUser_('', password, usuario, usuario);
  Logger.log('Login correcto: ' + (out && out.ok === true));
  if (out && !out.ok) Logger.log('Motivo: ' + out.message);
}
